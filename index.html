// server.js - Central Platform Engine with Unified Stripe Payment const express = require('express'); const stripe = require('stripe')(process.env.STRIPE_SECRET_KEY); const QRCode = require('qrcode'); // Erzeugt QR-Codes für physische Listings const app = express(); app.use(express.json()); // Liste der Ziel-Plattformen const SUPPORTED_PLATFORMS = [ 'AMAZON', 'EBAY', 'KAUFLAND', 'OTTO', 'ALIEXPRESS', 'ETSY', 'WALMART' ]; /** * 1. Generiert eine spezifische Stripe Checkout-Session für jede Plattform */ app.post('/api/generate-platform-link', async (req, res) => { const { productId, productName, amountCents, platform } = req.body; const targetPlatform = platform ? platform.toUpperCase() : 'DIRECT'; try { const session = await stripe.checkout.sessions.create({ payment_method_types: ['card', 'paypal', 'klarna', 'sepa_debit', 'giropay'], line_items: [ { price_data: { currency: 'eur', product_data: { name: productName, description: `Produkt-ID: ${productId} | Kanal: ${targetPlatform}`, }, unit_amount: amountCents, // z.B. 4999 = 49,99 € }, quantity: 1, }, ], mode: 'payment', // Erfolg- und Abbruch-URLs auf eurer eigenen Server/Kiosk-Domain success_url: `https://eure-webseite.de/success?session_id={CHECKOUT_SESSION_ID}&platform=${targetPlatform}`, cancel_url: `https://eure-webseite.de/cancel`, metadata: { product_id: productId, vendor_platform: targetPlatform, }, }); // Generiert zusätzlich einen QR-Code für Marktplatz-Bilder / Inserate const qrCodeDataUrl = await QRCode.toDataURL(session.url); res.json({ success: true, platform: targetPlatform, stripeCheckoutUrl: session.url, qrCodeBase64: qrCodeDataUrl }); } catch (error) { res.status(500).json({ success: false, error: error.message }); } }); /** * 2. Multi-Plattform Bulk Link Generator * Erstellt auf einen Schlag Links für ALLE großen Firmen */ app.post('/api/generate-all-platforms', async (req, res) => { const { productId, productName, amountCents } = req.body; const platformLinks = {}; try { for (const p of SUPPORTED_PLATFORMS) { const session = await stripe.checkout.sessions.create({ payment_method_types: ['card', 'paypal', 'klarna'], line_items: [ { price_data: { currency: 'eur', product_data: { name: `${productName} (${p})` }, unit_amount: amountCents, }, quantity: 1, }, ], mode: 'payment', success_url: `https://desert-guard-mt9l3hb7h-mettotech.vercel.app/=${p}`, cancel_url: `https://buy.stripe.com/eVq00bajMaZ6g34abT5c400l`, metadata: { product_id: productId, vendor_platform: p }, }); platformLinks[p] = session.url; } res.json({ success: true, productId, links: platformLinks }); } catch (error) { res.status(500).json({ success: false, error: error.message }); } }); /** * 3. Stripe Webhook Handler - Kassiert das Geld ab & startet Leistung sofort */ app.post('/webhook', express.raw({ type: 'application/json' }), (req, res) => { const sig = req.headers['stripe-signature']; let event; try { event = stripe.webhooks.constructEvent(req.body, sig, process.env.STRIPE_WEBHOOK_SECRET); } catch (err) { return res.status(400).send(`Webhook Error: ${err.message}`); } if (event.type === 'checkout.session.completed') { const session = event.data.object; const platform = session.metadata.vendor_platform; const productId = session.metadata.product_id; console.log(`💰 GELD EMPFANGEN ÜBER STRIPE!`); console.log(`► Quelle: ${platform}`); console.log(`► Produkt-ID: ${productId}`); console.log(`► Summe: ${session.amount_total / 100} €`); // HIER: Eure Freischalt- / Versand- / Kiosk-Logik ausführen processOrderDelivery(productId, platform, session.customer_details); } res.json({ received: true }); }); function processOrderDelivery(productId, platform, customerDetails) { // Automatische Bereitstellung (Freischaltung / Ticket / Versand) console.log(`🚀 Leistung für ${customerDetails.email} sofort aktiviert!`); } const PORT = process.env.PORT || 4242; app.listen(PORT, () => console.log(`🚀 Multi-Platform Stripe Engine läuft auf Port ${PORT}`));
